**Goal:** Add an Insights page (Usage tab) that shows words, speed, dictations, streak, notes, meetings, cleanup fixes, a 52-week heatmap and per-app / per-style use, from a local `insights.json` that survives history deletion.

```swift
    /// Word-level Levenshtein distance. Case and punctuation are ignored.
    public static func editDistance(reference: String, hypothesis: String) -> Int {
        let ref = words(reference)
        let hyp = words(hypothesis)
        if ref.isEmpty { return hyp.count }
        if hyp.isEmpty { return ref.count }
        var previous = Array(0...hyp.count)
        for i in 1...ref.count {
            var current = [i] + Array(repeating: 0, count: hyp.count)
            for j in 1...hyp.count {
                let cost = ref[i - 1] == hyp[j - 1] ? 0 : 1
                current[j] = min(previous[j] + 1, current[j - 1] + 1, previous[j - 1] + cost)
            }
            previous = current
        }
        return previous[hyp.count]
    }

    public static func compute(reference: String, hypothesis: String) -> Double {
        let referenceCount = words(reference).count
        if referenceCount == 0 { return words(hypothesis).isEmpty ? 0 : 1 }
        return Double(editDistance(reference: reference, hypothesis: hypothesis)) / Double(referenceCount)
    }
```

```swift
import Foundation

/// One day's activity. Numbers, app and style names and note IDs only:
/// never dictated text.
public struct InsightsDay: Codable, Equatable, Sendable {
    public var words = 0
    public var dictations = 0
    public var secondsSpoken: Double = 0
    public var notes = 0
    public var noteIDs: [String] = []
    public var noteWords = 0
    public var meetings = 0
    public var meetingSeconds: Double = 0
    public var wordsByApp: [String: Int] = [:]
    public var wordsByStyle: [String: Int] = [:]
    public var cleanupFixes = 0

    public init() {}

    /// Missing keys decode as zero, so later versions can add fields.
    public init(from decoder: any Decoder) throws {
        let c = try decoder.container(keyedBy: CodingKeys.self)
        words = try c.decodeIfPresent(Int.self, forKey: .words) ?? 0
        dictations = try c.decodeIfPresent(Int.self, forKey: .dictations) ?? 0
        secondsSpoken = try c.decodeIfPresent(Double.self, forKey: .secondsSpoken) ?? 0
        notes = try c.decodeIfPresent(Int.self, forKey: .notes) ?? 0
        noteIDs = try c.decodeIfPresent([String].self, forKey: .noteIDs) ?? []
        noteWords = try c.decodeIfPresent(Int.self, forKey: .noteWords) ?? 0
        meetings = try c.decodeIfPresent(Int.self, forKey: .meetings) ?? 0
        meetingSeconds = try c.decodeIfPresent(Double.self, forKey: .meetingSeconds) ?? 0
        wordsByApp = try c.decodeIfPresent([String: Int].self, forKey: .wordsByApp) ?? [:]
        wordsByStyle = try c.decodeIfPresent([String: Int].self, forKey: .wordsByStyle) ?? [:]
        cleanupFixes = try c.decodeIfPresent(Int.self, forKey: .cleanupFixes) ?? 0
    }

    /// A day counts toward a streak when anything was spoken or recorded.
    public var hasActivity: Bool { words > 0 || noteWords > 0 || notes > 0 || meetings > 0 }

    public mutating func merge(_ other: InsightsDay) {
        words += other.words
        dictations += other.dictations
        secondsSpoken += other.secondsSpoken
        noteWords += other.noteWords
        meetings += other.meetings
        meetingSeconds += other.meetingSeconds
        cleanupFixes += other.cleanupFixes
        for (app, count) in other.wordsByApp { wordsByApp[app, default: 0] += count }
        for (style, count) in other.wordsByStyle { wordsByStyle[style, default: 0] += count }
        for id in other.noteIDs where !noteIDs.contains(id) { noteIDs.append(id) }
        notes = noteIDs.count
    }
}

/// `insights.json`.
public struct InsightsFile: Codable, Equatable, Sendable {
    public static let supportedSchemaVersion = 1

    public var schemaVersion = InsightsFile.supportedSchemaVersion
    public var backfilledFromHistory = false
    /// Keyed by local day, `yyyy-MM-dd`.
    public var days: [String: InsightsDay] = [:]

    public init() {}

    public init(from decoder: any Decoder) throws {
        let c = try decoder.container(keyedBy: CodingKeys.self)
        schemaVersion = try c.decodeIfPresent(Int.self, forKey: .schemaVersion) ?? Self.supportedSchemaVersion
        backfilledFromHistory = try c.decodeIfPresent(Bool.self, forKey: .backfilledFromHistory) ?? false
        days = try c.decodeIfPresent([String: InsightsDay].self, forKey: .days) ?? [:]
    }

    public mutating func add(_ delta: InsightsDelta, at date: Date, calendar: Calendar) {
        days[InsightsDayKey.key(for: date, calendar: calendar), default: InsightsDay()].merge(delta.day)
    }
}

public enum InsightsDayKey {
    public static func key(for date: Date, calendar: Calendar) -> String {
        let parts = calendar.dateComponents([.year, .month, .day], from: date)
        return String(format: "%04d-%02d-%02d", parts.year ?? 0, parts.month ?? 0, parts.day ?? 0)
    }

    /// Midnight of that day in `calendar`'s time zone; nil for a malformed key.
    public static func date(for key: String, calendar: Calendar) -> Date? {
        let parts = key.split(separator: "-").compactMap { Int($0) }
        guard parts.count == 3 else { return nil }
        return calendar.date(from: DateComponents(year: parts[0], month: parts[1], day: parts[2]))
    }
}

/// What one event adds to its day.
public struct InsightsDelta: Equatable, Sendable {
    public var day: InsightsDay

    public init(day: InsightsDay) {
        self.day = day
    }

    public static func wordCount(_ text: String) -> Int {
        text.split(whereSeparator: { $0.isWhitespace || $0.isNewline }).count
    }

    /// Nil when nothing counts: secure fields, failures, empty text.
    public static func dictation(_ result: DictationResult, noteID: NoteID?) -> InsightsDelta? {
        spoken(outcome: result.outcome, appName: result.appName, style: result.style, raw: result.raw,
               cleaned: result.cleaned, text: result.finalText, seconds: result.timings.recording, noteID: noteID)
    }

    public static func history(_ entry: HistoryEntry) -> InsightsDelta? {
        guard let outcome = DictationOutcome(rawValue: entry.outcome) else { return nil }
        return spoken(outcome: outcome, appName: entry.appName, style: entry.style, raw: entry.raw,
                      cleaned: entry.cleaned, text: entry.text, seconds: entry.timings.recording, noteID: nil)
    }

    public static func meeting(seconds: TimeInterval) -> InsightsDelta? {
        guard seconds > 0 else { return nil }
        var day = InsightsDay()
        day.meetings = 1
        day.meetingSeconds = seconds
        return InsightsDelta(day: day)
    }

    private static func spoken(
        outcome: DictationOutcome, appName: String?, style: Style, raw: String, cleaned: String?,
        text: String, seconds: Double, noteID: NoteID?
    ) -> InsightsDelta? {
        let words = wordCount(text)
        guard words > 0 else { return nil }
        var day = InsightsDay()
        switch outcome {
        case .inserted, .copiedNotConsumed:
            let app = appName.flatMap { $0.isEmpty ? nil : $0 } ?? "Unknown"
            day.words = words
            day.wordsByApp[app] = words
            day.wordsByStyle[style.rawValue] = words
        case .insertedInNote:
            day.noteWords = words
            if let noteID {
                day.noteIDs = [noteID.rawValue]
                day.notes = 1
            }
        case .copiedSecureField, .discardedTooShort, .nothingHeard, .transcriptionFailed, .cancelled, .noteDeleted, .noteSaveFailed:
            return nil
        }
        day.dictations = 1
        day.secondsSpoken = max(0, seconds)
        day.cleanupFixes = cleaned.map { WordErrorRate.editDistance(reference: raw, hypothesis: $0) } ?? 0
        return InsightsDelta(day: day)
    }
}
```

```swift
import Foundation

public enum InsightsLoadIssue: Equatable, Sendable {
    case none
    /// The file could not be read; it was renamed aside and Insights starts
    /// empty, to be refilled from history.
    case rebuiltFromCorrupt
    /// Written by a newer BugVoice; left untouched and never written.
    case newerVersion
}

public enum InsightsStoreError: Error, Equatable {
    case newerVersion
}

/// Owns `insights.json`. Every write replaces the file atomically.
public actor InsightsStore {
    public let url: URL
    private var file = InsightsFile()
    private var loaded = false
    private var readOnly = false

    public init(url: URL) {
        self.url = url
    }

    @discardableResult
    public func load(now: Date = Date()) -> (InsightsFile, InsightsLoadIssue) {
        loaded = true
        readOnly = false
        file = InsightsFile()
        guard FileManager.default.fileExists(atPath: url.path) else { return (file, .none) }
        struct Header: Decodable { var schemaVersion: Int? }
        if let data = try? Data(contentsOf: url) {
            if let version = (try? JSONDecoder().decode(Header.self, from: data))?.schemaVersion,
               version > InsightsFile.supportedSchemaVersion {
                readOnly = true
                return (file, .newerVersion)
            }
            if let decoded = try? JSONDecoder().decode(InsightsFile.self, from: data) {
                file = decoded
                return (file, .none)
            }
        }
        let backup = url.deletingLastPathComponent()
            .appendingPathComponent("insights.corrupt-\(Int(now.timeIntervalSince1970)).json")
        try? FileManager.default.moveItem(at: url, to: backup)
        return (file, .rebuiltFromCorrupt)
    }

    public func add(_ delta: InsightsDelta, at date: Date, calendar: Calendar = .current) throws -> InsightsFile {
        try ensureWritable()
        var next = file
        next.add(delta, at: date, calendar: calendar)
        try commit(next)
        return file
    }

    /// Adds every history entry once. Later calls return the file unchanged.
    public func backfillIfNeeded(from entries: [HistoryEntry], calendar: Calendar = .current) throws -> InsightsFile {
        try ensureWritable()
        guard !file.backfilledFromHistory else { return file }
        var next = file
        for entry in entries {
            if let delta = InsightsDelta.history(entry) { next.add(delta, at: entry.startedAt, calendar: calendar) }
        }
        next.backfilledFromHistory = true
        try commit(next)
        return file
    }

    /// Back to zero. Marks the backfill as done so history does not refill it.
    public func reset() throws -> InsightsFile {
        try ensureWritable()
        var next = InsightsFile()
        next.backfilledFromHistory = true
        try commit(next)
        return file
    }

    private func ensureWritable() throws {
        if !loaded { load() }
        if readOnly { throw InsightsStoreError.newerVersion }
    }

    private func commit(_ next: InsightsFile) throws {
        try FileManager.default.createDirectory(at: url.deletingLastPathComponent(), withIntermediateDirectories: true)
        let encoder = JSONEncoder()
        encoder.outputFormatting = [.prettyPrinted, .sortedKeys]
        try encoder.encode(next).write(to: url, options: .atomic)
        file = next
    }
}
```

```swift
/// Fixed thresholds so a colour means the same amount every month.
public enum HeatLevel {
    public static func level(forWords words: Int) -> Int {
        switch words {
        case ..<1: 0
        case 1..<150: 1
        case 150..<300: 2
        case 300..<480: 3
        default: 4
        }
    }
}

/// Everything the Insights Usage tab shows.
public struct InsightsSummary: Equatable, Sendable {
    public var words = 0
    public var wordsToday = 0
    public var wordsPerMinute = 0
    public var dictations = 0
    public var currentStreak = 0
    public var longestStreak = 0
    public var notes = 0
    public var noteWords = 0
    public var meetings = 0
    public var meetingHours: Double = 0
    public var cleanupFixes = 0
    public var appCount = 0
    public var apps: [BreakdownRow] = []
    public var styles: [BreakdownRow] = []
    public var weeks: [HeatWeek] = []

    public static let weekCount = 52
    public static let topAppCount = 5

    public static func compute(file: InsightsFile, period: InsightsPeriod, today: Date, calendar: Calendar = .current) -> InsightsSummary {
        let start = calendar.startOfDay(for: today)
        func shifted(_ days: Int, from date: Date = start) -> Date {
            calendar.date(byAdding: .day, value: days, to: date) ?? date
        }
        func key(_ date: Date) -> String { InsightsDayKey.key(for: date, calendar: calendar) }

        let periodKeys: Set<String>? = period.dayCount.map { count in Set((0..<count).map { key(shifted(-$0)) }) }
        func inPeriod(_ dayKey: String) -> Bool { periodKeys?.contains(dayKey) ?? true }

        var summary = InsightsSummary()
        var seconds = 0.0
        var meetingSeconds = 0.0
        var apps: [String: Int] = [:]
        var styles: [String: Int] = [:]
        for (dayKey, day) in file.days where inPeriod(dayKey) {
            summary.words += day.words
            summary.dictations += day.dictations
            summary.notes += day.notes
            summary.noteWords += day.noteWords
            summary.meetings += day.meetings
            summary.cleanupFixes += day.cleanupFixes
            seconds += day.secondsSpoken
            meetingSeconds += day.meetingSeconds
            for (app, count) in day.wordsByApp { apps[app, default: 0] += count }
            for (style, count) in day.wordsByStyle { styles[style, default: 0] += count }
        }
        summary.wordsToday = file.days[key(start)]?.words ?? 0
        summary.wordsPerMinute = seconds > 0 ? Int((Double(summary.words + summary.noteWords) / seconds * 60).rounded()) : 0
        summary.meetingHours = (meetingSeconds / 3600 * 10).rounded() / 10

        // Streaks are all-time: the period never changes them.
        func active(_ date: Date) -> Bool { file.days[key(date)]?.hasActivity ?? false }
        var cursor = active(start) ? start : shifted(-1)
        while active(cursor) {
            summary.currentStreak += 1
            cursor = shifted(-1, from: cursor)
        }
        let activeDates = file.days.filter { $0.value.hasActivity }
            .compactMap { InsightsDayKey.date(for: $0.key, calendar: calendar) }
            .sorted()
        var run = 0
        var previous: Date?
        for date in activeDates {
            run = previous.map { shifted(1, from: $0) == date } == true ? run + 1 : 1
            summary.longestStreak = max(summary.longestStreak, run)
            previous = date
        }

        let appTotal = apps.values.reduce(0, +)
        let sortedApps = apps.sorted { $0.value != $1.value ? $0.value > $1.value : $0.key < $1.key }
        summary.appCount = sortedApps.count
        func share(_ value: Int, of total: Int) -> Double { total > 0 ? Double(value) / Double(total) : 0 }
        summary.apps = sortedApps.prefix(topAppCount).map { BreakdownRow(name: $0.key, words: $0.value, share: share($0.value, of: appTotal)) }
        if sortedApps.count > topAppCount {
            let rest = sortedApps.dropFirst(topAppCount).reduce(0) { $0 + $1.value }
            summary.apps.append(BreakdownRow(name: "Other", words: rest, share: share(rest, of: appTotal)))
        }
        let styleTotal = styles.values.reduce(0, +)
        summary.styles = styles.filter { $0.value > 0 }
            .sorted { $0.value != $1.value ? $0.value > $1.value : $0.key < $1.key }
            .map { BreakdownRow(name: $0.key, words: $0.value, share: share($0.value, of: styleTotal)) }

        let weekday = calendar.component(.weekday, from: start)  // 1 = Sunday
        let firstSunday = shifted(-(weekday - 1) - 7 * (weekCount - 1))
        let monthNames = calendar.shortStandaloneMonthSymbols
        var lastMonth: Int?
        for column in 0..<weekCount {
            let weekStart = shifted(column * 7, from: firstSunday)
            var days: [HeatCell?] = []
            for offset in 0..<7 {
                let date = shifted(offset, from: weekStart)
                guard date <= start else { days.append(nil); continue }
                let dayKey = key(date)
                let day = file.days[dayKey]
                let words = (day?.words ?? 0) + (day?.noteWords ?? 0)
                days.append(HeatCell(key: dayKey, date: date, words: words, dictations: day?.dictations ?? 0,
                                     level: HeatLevel.level(forWords: words), inPeriod: inPeriod(dayKey)))
            }
            let month = calendar.component(.month, from: weekStart)
            let label = column > 0 && month != lastMonth ? monthNames[month - 1] : nil
            lastMonth = month
            summary.weeks.append(HeatWeek(id: column, days: days, monthLabel: label))
        }
        return summary
    }
}
```

```swift
import AppKit

/// Rounded segmented control: a capsule track with the selected pill raised.
/// Used for page tabs and small filters.
struct PillTabs<Value: Hashable>: View {
    struct Item {
        var value: Value
        var title: String
        var badge: String? = nil
        var isEnabled = true
    }

    enum Size { case regular, small }

    let items: [Item]
    @Binding var selection: Value
    var size: Size = .regular

    var body: some View {
        HStack(spacing: 2) {
            ForEach(items.indices, id: \.self) { index in
                pill(items[index])
            }
        }
        .padding(3)
        .background(Capsule().fill(Color.primary.opacity(0.07)))
    }

    private func pill(_ item: Item) -> some View {
        let selected = item.value == selection
        return Button {
            selection = item.value
        } label: {
            HStack(spacing: 5) {
                Text(item.title)
                if let badge = item.badge {
                    Text(badge)
                        .font(.system(size: 9, weight: .semibold))
                        .padding(.horizontal, 4).padding(.vertical, 1)
                        .background(Capsule().fill(Color.primary.opacity(0.08)))
                }
            }
            .font(size == .small ? .callout : .body)
            .fontWeight(selected ? .semibold : .regular)
            .foregroundStyle(selected ? Color.primary : Color.secondary)
            .padding(.horizontal, size == .small ? 10 : 14)
            .padding(.vertical, size == .small ? 3 : 5)
            .background {
                if selected {
                    Capsule()
                        .fill(Color(nsColor: .controlBackgroundColor))
                        .shadow(color: .black.opacity(0.12), radius: 1, y: 1)
                }
            }
            .contentShape(Capsule())
        }
        .buttonStyle(.plain)
        .disabled(!item.isEnabled)
        .opacity(item.isEnabled ? 1 : 0.55)
        .accessibilityAddTraits(selected ? .isSelected : [])
    }
}
```

```swift
                    RoundedRectangle(cornerRadius: 3).strokeBorder(Color.primary, lineWidth: 1.5)
                }
            }
            .onHover { inside in
                if inside { hovered = day } else if hovered == day { hovered = nil }
            }
```
