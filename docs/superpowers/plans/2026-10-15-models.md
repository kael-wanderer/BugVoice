**Goal:** Replace the single `textModel` setting with connections (each with its own Keychain key) and a "Use for" choice per feature — Dictation cleanup, Notes dictation, Meeting notes — including `claude` / `codex` / `pi` subprocesses for Meeting notes.

The app turns a `ModelPlan` into a `TextModel` in `ProviderFactory`, moves old Keychain keys once, and gets a new Settings → Models tab.

**Tech Stack:** Swift 6.4, tools 6.0, macOS 15, SwiftUI + AppKit, Foundation `Process`, swift-testing. No new dependencies.

## Review Focus

1. **Upgrade keeps dictation working.** An existing `settings.json` with `textModel` (LM Studio, key under `openai`) loads as one connection used by Dictation and Meeting notes; the old key is copied once to `connection-migrated-1`; the next save drops `textModel`. Pinned by Task 1 migration tests and Task 6 Step 4 (key move in `bootstrap`).
2. **A CLI that never ends.** Cancel or timeout must stop the subprocess and return, even when it writes more than a pipe buffer or ignores stdin. Pinned by Task 5 `FoundationProcessRunner` tests (timeout, cancellation, 200 KB output).
3. **Key sent somewhere unexpected.** Plain `http://` to a LAN host, a URL with user:password, and a 3xx redirect never carry a key. Pinned by Task 2 address tests and Task 4 `redirectIsAnErrorNotFollowed` / `noRedirectDelegateRefuses`.
4. **OpenAI reasoning models answer empty.** `gpt-5*` spend `max_completion_tokens` on reasoning; a dictation budget of ~60 tokens returns nothing. The OpenAI flavor sends `max(maxTokens, 2048)`. Pinned by Task 4 `openAIFlavorBody`.
5. **A deleted connection still selected.** Deleting the connection Dictation uses must give raw text with a clear reason, not a crash or a silent skip. Pinned by Task 3 `deletedConnectionIsUnavailable` and Task 6's `UnavailableModel` mapping.

```swift
import Foundation

public enum ConnectionKind: String, Codable, Sendable, CaseIterable {
    case openAICompatible, anthropicCompatible, openAI, anthropic, openRouter

    public var title: String {
        switch self {
        case .openAICompatible: "OpenAI-compatible"
        case .anthropicCompatible: "Anthropic-compatible"
        case .openAI: "OpenAI"
        case .anthropic: "Anthropic"
        case .openRouter: "OpenRouter"
        }
    }

    public var usesAnthropicAPI: Bool { self == .anthropic || self == .anthropicCompatible }
    public var isCloud: Bool { self == .openAI || self == .anthropic || self == .openRouter }

    var idPrefix: String {
        switch self {
        case .openAICompatible: "server"
        case .anthropicCompatible: "anthropic-server"
        case .openAI: "openai"
        case .anthropic: "anthropic"
        case .openRouter: "openrouter"
        }
    }
}

/// Coding CLIs that can write meeting notes. Gemini is out of scope.
public enum CLITool: String, Codable, Sendable, CaseIterable {
    case claude, codex, pi

    public var title: String {
        switch self {
        case .claude: "Claude Code"
        case .codex: "Codex"
        case .pi: "Pi"
        }
    }
}

/// A model server. Its API key is in Keychain under `keychainAccount`.
public struct ModelConnection: Codable, Equatable, Sendable, Identifiable {
    public var id: String
    public var name: String
    public var kind: ConnectionKind
    /// As typed. Ignored for kinds with a fixed address.
    public var baseURL: String
    /// Default model id; a feature may override it.
    public var model: String

    public init(id: String, name: String, kind: ConnectionKind, baseURL: String, model: String) {
        self.id = id
        self.name = name
        self.kind = kind
        self.baseURL = baseURL
        self.model = model
    }

    public var keychainAccount: String { "connection-\(id)" }

    /// Lowercase ASCII letters, digits and `-`; never `cli-…`.
    public static func isValidID(_ id: String) -> Bool {
        !id.isEmpty && id.count <= 64 && !id.hasPrefix("cli-")
            && id.allSatisfy { $0 == "-" || ($0.isASCII && ($0.isLowercase || $0.isNumber)) }
    }

    public static func newID(for kind: ConnectionKind) -> String {
        kind.idPrefix + "-" + UUID().uuidString.prefix(6).lowercased()
    }
}

extension ModelConnection {
    public init(from decoder: Decoder) throws {
        let c = try decoder.container(keyedBy: CodingKeys.self)
        id = try c.decode(String.self, forKey: .id)
        kind = try c.decode(ConnectionKind.self, forKey: .kind)
        name = c.value(.name, default: kind.title)
        baseURL = c.value(.baseURL, default: "")
        model = c.value(.model, default: "")
    }
}

/// How a feature reaches a model.
public enum ModelEngine: Codable, Equatable, Sendable, Hashable {
    case connection(String)
    case cli(CLITool)
    case appleIntelligence
    case none

    private enum Keys: String, CodingKey { case connection, cli }

    public init(from decoder: Decoder) throws {
        if let single = try? decoder.singleValueContainer().decode(String.self) {
            switch single {
            case "appleIntelligence": self = .appleIntelligence
            case "none": self = .none
            default:
                throw DecodingError.dataCorrupted(.init(codingPath: decoder.codingPath, debugDescription: "Unknown engine \(single)"))
            }
            return
        }
        let c = try decoder.container(keyedBy: Keys.self)
        if let id = try c.decodeIfPresent(String.self, forKey: .connection) {
            self = .connection(id)
        } else {
            self = .cli(try c.decode(CLITool.self, forKey: .cli))
        }
    }

    public func encode(to encoder: Encoder) throws {
        switch self {
        case .connection(let id):
            var c = encoder.container(keyedBy: Keys.self)
            try c.encode(id, forKey: .connection)
        case .cli(let tool):
            var c = encoder.container(keyedBy: Keys.self)
            try c.encode(tool, forKey: .cli)
        case .appleIntelligence:
            var c = encoder.singleValueContainer()
            try c.encode("appleIntelligence")
        case .none:
            var c = encoder.singleValueContainer()
            try c.encode("none")
        }
    }
}

/// One feature's choice. An empty model means the connection's default; an
/// empty effort means the CLI's default.
public struct ModelUse: Codable, Equatable, Sendable {
    public var engine: ModelEngine
    public var model: String
    /// API engines: ask the server to skip reasoning.
    public var disableThinking: Bool
    /// CLI engines only.
    public var effort: String

    public init(engine: ModelEngine = .none, model: String = "", disableThinking: Bool = false, effort: String = "") {
        self.engine = engine
        self.model = model
        self.disableThinking = disableThinking
        self.effort = effort
    }
}

extension ModelUse {
    public init(from decoder: Decoder) throws {
        self.init()
        let c = try decoder.container(keyedBy: CodingKeys.self)
        engine = c.value(.engine, default: engine)
        model = c.value(.model, default: model)
        disableThinking = c.value(.disableThinking, default: disableThinking)
        effort = c.value(.effort, default: effort)
    }

    /// CLIs are too slow for dictation.
    func withoutCLI() -> ModelUse {
        guard case .cli = engine else { return self }
        var copy = self
        copy.engine = .none
        return copy
    }
}

public struct ModelSettings: Codable, Equatable, Sendable {
    public static let maxConnections = 20
    public static let defaultConnection = ModelConnection(
        id: "lmstudio", name: "LM Studio", kind: .openAICompatible,
        baseURL: "http://localhost:1234", model: "qwen3.8-27b-mlx"
    )

    public var connections: [ModelConnection] = [ModelSettings.defaultConnection]
    public var dictation = ModelUse(engine: .connection("lmstudio"), disableThinking: true)
    /// Nil: same as Dictation.
    public var notes: ModelUse?
    public var meetings = ModelUse(engine: .connection("lmstudio"))
    /// Dictation and Notes: after this, raw text is pasted.
    public var dictationTimeoutSeconds: Double = 3
    /// Set by migration, done by the app at launch, never saved.
    public var pendingKeyMoves: [KeyMove] = []

    public init() {}

    enum CodingKeys: String, CodingKey {
        case connections, dictation, notes, meetings, dictationTimeoutSeconds
    }
}

extension ModelSettings {
    public init(from decoder: Decoder) throws {
        self.init()
        let c = try decoder.container(keyedBy: CodingKeys.self)
        if c.contains(.connections) {
            var seen = Set<String>()
            connections = Array(
                c.value(.connections, default: [LossyElement<ModelConnection>]())
                    .compactMap(\.value)
                    .filter { ModelConnection.isValidID($0.id) && seen.insert($0.id).inserted }
                    .prefix(Self.maxConnections)
            )
        }
        dictation = c.value(.dictation, default: dictation).withoutCLI()
        notes = c.value(.notes, default: notes)?.withoutCLI()
        meetings = c.value(.meetings, default: meetings)
        dictationTimeoutSeconds = c.value(.dictationTimeoutSeconds, default: dictationTimeoutSeconds)
    }
}

/// `textModel` (one provider, one key) → `ModelSettings`.
public enum LegacyTextModelMigration {
    public static func migrate(_ old: TextModelSettings) -> ModelSettings {
        var s = ModelSettings()
        s.dictationTimeoutSeconds = old.timeoutSeconds
        let engine: ModelEngine
        switch old.provider {
        case .none:
            s.connections = []
            engine = .none
        case .appleIntelligence:
            s.connections = []
            engine = .appleIntelligence
        case .openAICompatible:
            let name = ConnectionPreset.all
                .first { $0.group == .local && $0.baseURL == old.baseURL }?.name ?? "OpenAI-compatible server"
            s.connections = [ModelConnection(id: "migrated-1", name: name, kind: .openAICompatible, baseURL: old.baseURL, model: old.model)]
            s.pendingKeyMoves = [KeyMove(from: "openai", to: "connection-migrated-1")]
            engine = .connection("migrated-1")
        case .anthropicCompatible:
            let official = URL(string: old.baseURL)?.host?.lowercased() == "api.anthropic.com"
            s.connections = [ModelConnection(
                id: "migrated-1", name: official ? "Anthropic" : "Anthropic-compatible server",
                kind: official ? .anthropic : .anthropicCompatible, baseURL: old.baseURL, model: old.model
            )]
            s.pendingKeyMoves = [KeyMove(from: "anthropic", to: "connection-migrated-1")]
            engine = .connection("migrated-1")
        }
        s.dictation = ModelUse(engine: engine, disableThinking: old.disableThinking)
        s.meetings = ModelUse(engine: engine)
        s.notes = nil
        return s
    }
}

public enum ModelNames {
    /// A short label for the HUD, e.g. *Cleaning (Qwen)*: the model id's
    /// leading word, capitalized, without an organization prefix.
    public static func short(_ modelID: String) -> String? {
        let id = modelID.trimmingCharacters(in: .whitespaces)
        guard !id.isEmpty else { return nil }
        let base = id.split(separator: "/").last.map(String.init) ?? id
        let word = base.prefix { $0.isLetter }
        guard !word.isEmpty else { return id }
        return word.prefix(1).uppercased() + word.dropFirst()
    }
}
```

```swift
import Foundation

/// Which address a connection may be called at. A key only ever goes to a
/// fixed cloud address, an https server, or this Mac.
public enum ConnectionAddress {
    public static func fixedURL(for kind: ConnectionKind) -> URL? {
        switch kind {
        case .openAI: URL(string: "https://api.openai.com")
        case .anthropic: URL(string: "https://api.anthropic.com")
        case .openRouter: URL(string: "https://openrouter.ai/api")
        case .openAICompatible, .anthropicCompatible: nil
        }
    }

    /// The address to call, or nil when the typed one breaks a rule.
    public static func baseURL(for connection: ModelConnection) -> URL? {
        if let fixed = fixedURL(for: connection.kind) { return fixed }
        guard problem(connection.baseURL) == nil else { return nil }
        return URL(string: connection.baseURL.trimmingCharacters(in: .whitespaces))
    }

    /// Nil when the address is usable; otherwise the sentence to show.
    public static func problem(_ typed: String) -> String? {
        let text = typed.trimmingCharacters(in: .whitespaces)
        guard let parts = URLComponents(string: text),
              let scheme = parts.scheme?.lowercased(), scheme == "http" || scheme == "https",
              let host = parts.host, !host.isEmpty
        else { return "Enter an address like http://localhost:1234." }
        if parts.user != nil || parts.password != nil {
            return "Remove the user name and password from the address."
        }
        if parts.query != nil || parts.fragment != nil {
            return "Remove the ? or # part of the address."
        }
        if scheme == "http", !isLoopback(host: host) {
            return "Use https:// for a server that is not on this Mac."
        }
        return nil
    }

    /// The host itself, not a name that merely starts with "localhost".
    public static func isLoopback(host: String) -> Bool {
        let h = host.lowercased().trimmingCharacters(in: CharacterSet(charactersIn: "[]"))
        return h == "localhost" || h == "127.0.0.1" || h == "::1"
    }

    public static func isLocal(_ url: URL) -> Bool {
        url.host.map { isLoopback(host: $0) } ?? false
    }
}
```

```swift
import Foundation

public enum ModelFeature: Sendable, CaseIterable {
    case dictation, notes, meetings

    public var title: String {
        switch self {
        case .dictation: "Dictation cleanup"
        case .notes: "Notes dictation"
        case .meetings: "Meeting notes"
        }
    }

    /// Dictation and Notes wait for the answer before pasting.
    var isFast: Bool { self != .meetings }
}

/// What a feature will call, decided from settings alone.
public enum ModelPlan: Equatable, Sendable {
    case none
    case unavailable(String)
    case connection(ModelConnection, baseURL: URL, modelID: String, disableThinking: Bool, timeout: TimeInterval)
    case appleIntelligence(timeout: TimeInterval)
    case cli(CLITool, path: String, model: String, effort: String, timeout: TimeInterval)

    /// For the HUD's *Cleaning (…)*.
    public var shortName: String? {
        switch self {
        case .none, .unavailable: nil
        case .connection(let c, _, let modelID, _, _): ModelNames.short(modelID) ?? c.name
        case .appleIntelligence: "Apple Intelligence"
        case .cli(let tool, _, _, _, _): tool.title
        }
    }
}

public enum ModelResolver {
    static let cloudTimeout: TimeInterval = 120
    static let localTimeout: TimeInterval = 600
    static let cliTimeout: TimeInterval = 600

    public static func plan(
        for feature: ModelFeature,
        settings: ModelSettings,
        cliPaths: [CLITool: String],
        appleIntelligenceAvailable: Bool
    ) -> ModelPlan {
        let use: ModelUse = switch feature {
        case .dictation: settings.dictation
        case .notes: settings.notes ?? settings.dictation
        case .meetings: settings.meetings
        }
        switch use.engine {
        case .none:
            return .none
        case .appleIntelligence:
            guard appleIntelligenceAvailable else { return .unavailable("Apple Intelligence is not available on this Mac") }
            return .appleIntelligence(timeout: feature.isFast ? settings.dictationTimeoutSeconds : cloudTimeout)
        case .cli(let tool):
            guard !feature.isFast else { return .none }
            guard let path = cliPaths[tool] else { return .unavailable("\(tool.rawValue) was not found on this Mac.") }
            return .cli(tool, path: path, model: use.model, effort: use.effort, timeout: cliTimeout)
        case .connection(let id):
            guard let connection = settings.connections.first(where: { $0.id == id }) else {
                return .unavailable("The connection for \(feature.title) was deleted. Choose another in Settings → Models.")
            }
            guard let url = ConnectionAddress.baseURL(for: connection) else {
                return .unavailable("The address of \(connection.name) is not valid.")
            }
            let override = use.model.trimmingCharacters(in: .whitespaces)
            let timeout = feature.isFast
                ? settings.dictationTimeoutSeconds
                : (ConnectionAddress.isLocal(url) ? localTimeout : cloudTimeout)
            return .connection(
                connection, baseURL: url, modelID: override.isEmpty ? connection.model : override,
                disableThinking: use.disableThinking, timeout: timeout
            )
        }
    }
}

    public var model: any TextModel
    public var modelID: String
    public var shortName: String?
    public var disableThinking: Bool
    public var timeoutSeconds: Double

    public init(model: any TextModel, modelID: String, shortName: String? = nil, disableThinking: Bool = false, timeoutSeconds: Double = 3) {
        self.model = model
        self.modelID = modelID
        self.shortName = shortName
        self.disableThinking = disableThinking
        self.timeoutSeconds = timeoutSeconds
    }
}

/// (meeting notes).
public struct ThinkingOverride: TextModel {
    public var base: any TextModel
    public var disableThinking: Bool

    public init(base: any TextModel, disableThinking: Bool) {
        self.base = base
        self.disableThinking = disableThinking
    }

    public func clean(_ request: CleanupRequest) async throws -> String {
        var request = request
        request.disableThinking = disableThinking
        return try await base.clean(request)
    }
}
```

```swift
/// Refuses every redirect: following one would re-send the key to whatever
/// host the response named.
public final class NoRedirectDelegate: NSObject, URLSessionTaskDelegate, Sendable {
    public func urlSession(
        _ session: URLSession, task: URLSessionTask,
        willPerformHTTPRedirection response: HTTPURLResponse, newRequest request: URLRequest
    ) async -> URLRequest? {
        nil
    }
}

enum HTTP {
    static let noRedirectSession = URLSession(configuration: .default, delegate: NoRedirectDelegate(), delegateQueue: nil)

    /// `redactBody`: drop the response body from errors when the request
    /// carried a key; providers echo the request back.
    static func send(_ request: URLRequest, session: URLSession, redactBody: Bool = false) async throws -> Data {
        let (data, response) = try await session.data(for: request)
        let status = (response as? HTTPURLResponse)?.statusCode ?? 0
        if (300..<400).contains(status) { throw ProviderError.redirected }
        guard (200..<300).contains(status) else {
            throw ProviderError.http(status: status, body: redactBody ? "" : String(decoding: data.prefix(2_000), as: UTF8.self))
        }
        return data
    }

    static func jsonRequest(url: URL, body: some Encodable, headers: [String: String], timeout: TimeInterval = 60) throws -> URLRequest {
        var request = URLRequest(url: url, timeoutInterval: timeout)
        request.httpMethod = "POST"
        request.setValue("application/json", forHTTPHeaderField: "Content-Type")
        for (key, value) in headers { request.setValue(value, forHTTPHeaderField: key) }
        request.httpBody = try JSONEncoder().encode(body)
        return request
    }
```

```swift
import Foundation

public struct ProcessResult: Equatable, Sendable {
    public var exitCode: Int32
    public var stdout: String
    public var stderr: String

    public init(exitCode: Int32, stdout: String, stderr: String) {
        self.exitCode = exitCode
        self.stdout = stdout
        self.stderr = stderr
    }
}

public enum ProcessRunnerError: Error, Equatable, LocalizedError {
    case timedOut
    case launchFailed(String)

    public var errorDescription: String? {
        switch self {
        case .timedOut: "The tool took too long and was stopped."
        case .launchFailed(let path): "Could not start \(path)."
        }
    }
}

public protocol ProcessRunner: Sendable {
    func run(executable: String, arguments: [String], stdin: String, workingDirectory: URL?, timeout: TimeInterval) async throws -> ProcessResult
}

/// Runs a subprocess with stdin, stdout and stderr through pipes. Output is
/// drained while the process runs (a full pipe would block it). Timeout and
/// task cancellation send SIGTERM, then SIGKILL after 2 s.
public struct FoundationProcessRunner: ProcessRunner {
    public init() {}

    public func run(executable: String, arguments: [String], stdin: String, workingDirectory: URL?, timeout: TimeInterval) async throws -> ProcessResult {
        let process = Process()
        process.executableURL = URL(fileURLWithPath: executable)
        process.arguments = arguments
        if let workingDirectory { process.currentDirectoryURL = workingDirectory }
        let input = Pipe(), output = Pipe(), errors = Pipe()
        process.standardInput = input
        process.standardOutput = output
        process.standardError = errors

        let out = PipeCollector(output.fileHandleForReading)
        let err = PipeCollector(errors.fileHandleForReading)
        let handle = ProcessHandle(process)
        let exit = AsyncStream.makeStream(of: Int32.self)
        process.terminationHandler = { finished in
            exit.continuation.yield(finished.terminationStatus)
            exit.continuation.finish()
        }
        do {
            try process.run()
        } catch {
            throw ProcessRunnerError.launchFailed(executable)
        }
        let writer = input.fileHandleForWriting
        let data = Data(stdin.utf8)
        Thread.detachNewThread {
            try? writer.write(contentsOf: data)
            try? writer.close()
        }
        let timer = Task {
            try? await Task.sleep(for: .seconds(timeout))
            if !Task.isCancelled { handle.stop(timedOut: true) }
        }
        let status = await withTaskCancellationHandler {
            var code: Int32 = -1
            for await value in exit.stream { code = value }
            return code
        } onCancel: {
            handle.stop(timedOut: false)
        }
        timer.cancel()
        let stdout = await out.finish()
        let stderr = await err.finish()
        if Task.isCancelled { throw CancellationError() }
        if handle.timedOut { throw ProcessRunnerError.timedOut }
        return ProcessResult(exitCode: status, stdout: stdout, stderr: stderr)
    }
}

/// Owns the process for stop requests from the timer and the cancel handler.
final class ProcessHandle: @unchecked Sendable {
    private let process: Process
    private let lock = NSLock()
    private var didTimeOut = false

    init(_ process: Process) { self.process = process }

    var timedOut: Bool { lock.withLock { didTimeOut } }

    func stop(timedOut: Bool) {
        lock.withLock { if timedOut { didTimeOut = true } }
        guard process.isRunning else { return }
        process.terminate()
        let pid = process.processIdentifier
        DispatchQueue.global().asyncAfter(deadline: .now() + 2) { [process] in
            if process.isRunning { kill(pid, SIGKILL) }
        }
    }
}

/// Collects a pipe as data arrives. `finish()` waits briefly for end of file,
/// since a child of the process may still hold the pipe open after it exits.
final class PipeCollector: @unchecked Sendable {
    private let lock = NSLock()
    private var data = Data()
    private var ended = false
    private let handle: FileHandle

    init(_ handle: FileHandle) {
        self.handle = handle
        handle.readabilityHandler = { [weak self] h in
            let chunk = h.availableData
            guard let self else { return }
            self.lock.withLock {
                if chunk.isEmpty { self.ended = true } else { self.data.append(chunk) }
            }
            if chunk.isEmpty { h.readabilityHandler = nil }
        }
    }

    func finish() async -> String {
        for _ in 0..<20 {
            if lock.withLock({ ended }) { break }
            try? await Task.sleep(for: .milliseconds(50))
        }
        handle.readabilityHandler = nil
        return lock.withLock { String(decoding: data, as: UTF8.self) }
    }
}
```

`finish()` waits at most 1 s for end of file after exit. If `cancellationStopsTheProcess` reports `CancellationError` not thrown because `Task.isCancelled` is false inside the child task, check cancellation with `try Task.checkCancellation()` instead of the `if`.

```swift
    }

                "-C", workingDirectory.path, "--output-last-message", answerFile(in: workingDirectory).path,

            return a

            if let effort { a += ["--thinking", effort] }

public enum CLIOutputError: Error, Equatable, LocalizedError {
    case noAnswer(CLITool, detail: String)
    case signedOut(CLITool)
    case toolFailed(CLITool)

    public var errorDescription: String? {
        switch self {
        case .noAnswer(let tool, let detail):
            detail.isEmpty ? "\(tool.rawValue) did not return an answer." : "\(tool.rawValue) did not return an answer. \(detail)"
        case .signedOut(let tool): "\(tool.rawValue) is signed out. Run \(tool.rawValue) in Terminal to sign in."
        case .toolFailed(let tool): "\(tool.rawValue) could not answer."
        }
    }
}

public enum CLIOutput {
    public static func answer(tool: CLITool, result: ProcessResult, codexAnswer: String?) throws -> String {
        switch tool {
        case .claude:
            if let object = json(result.stdout) {
                if object["is_error"] as? Bool == true { throw CLIOutputError.toolFailed(.claude) }
                if let text = (object["result"] as? String)?.trimmingCharacters(in: .whitespacesAndNewlines), !text.isEmpty {
                    return text
                }
            }
        case .codex:
            if result.exitCode == 0, let text = codexAnswer?.trimmingCharacters(in: .whitespacesAndNewlines), !text.isEmpty {
                return text
            }
        case .pi:
            var answer: String?
            for line in result.stdout.split(whereSeparator: \.isNewline) {
                guard let event = json(String(line)), let message = event["message"] as? [String: Any],
                      message["role"] as? String == "assistant", message["stopReason"] as? String != "error"
                else { continue }
                let text = (message["content"] as? [[String: Any]] ?? []).compactMap { $0["text"] as? String }.joined()
                    .trimmingCharacters(in: .whitespacesAndNewlines)
                if !text.isEmpty { answer = text }
            }
            if let answer { return answer }
        }
        let said = (result.stdout + " " + result.stderr).lowercased()
        if ["/login", "not logged in", "please log in", "codex login", "unauthorized"].contains(where: said.contains) {
            throw CLIOutputError.signedOut(tool)
        }
        throw CLIOutputError.noAnswer(tool, detail: String(result.stderr.trimmingCharacters(in: .whitespacesAndNewlines).suffix(300)))
    }

    static func json(_ text: String) -> [String: Any]? {
        (try? JSONSerialization.jsonObject(with: Data(text.trimmingCharacters(in: .whitespacesAndNewlines).utf8))) as? [String: Any]
    }
}

    public static func locate(_ tool: CLITool, runner: any ProcessRunner) async -> CLIInfo? {

            executable: "/bin/zsh", arguments: ["-lc", "command -v \(tool.rawValue)"], stdin: "", workingDirectory: nil, timeout: 10
        ), found.exitCode == 0,
            let path = found.stdout.split(whereSeparator: \.isNewline).map(String.init).last(where: { $0.hasPrefix("/") })
        else { return nil }
        let version = (try? await runner.run(executable: path, arguments: ["--version"], stdin: "", workingDirectory: nil, timeout: 10))?
            .stdout.split(whereSeparator: \.isNewline).first.map { $0.trimmingCharacters(in: .whitespaces) } ?? ""
        return CLIInfo(path: path, version: version)
```

```swift
import SwiftUI
```

```swift
struct ConnectionSheet: View {
    let model: AppModel
    let isNew: Bool
    let original: ModelConnection
    @State private var connection: ModelConnection
    @State private var key = ""
    @State private var keyRemoved = false
    @State private var models: [String] = []
    @State private var testResult: String?
    @State private var testing = false
    @Environment(\.dismiss) private var dismiss

    init(model: AppModel, draft: ConnectionDraft) {
        self.model = model
        isNew = draft.isNew
        original = draft.connection
        _connection = State(initialValue: draft.connection)
    }

    private var fixed: Bool { ConnectionAddress.fixedURL(for: connection.kind) != nil }
    private var addressProblem: String? { fixed ? nil : ConnectionAddress.problem(connection.baseURL) }
    private var addressChanged: Bool { !isNew && connection.baseURL.trimmingCharacters(in: .whitespaces) != original.baseURL.trimmingCharacters(in: .whitespaces) }
    private var hadKey: Bool { !(model.keychain.read(original.keychainAccount) ?? "").isEmpty }
    private var showsKey: Bool {
        connection.kind.isCloud || !(ConnectionAddress.baseURL(for: connection).map(ConnectionAddress.isLocal) ?? true)
            || hadKey
    }

    var body: some View {
        VStack(alignment: .leading, spacing: 12) {
            Text(isNew ? "Add connection" : "Edit connection").font(.headline)
            Form {
                TextField("Name", text: $connection.name)
                LabeledContent("Kind", value: connection.kind.title)
                if fixed {
                    LabeledContent("Address", value: ConnectionAddress.fixedURL(for: connection.kind)?.absoluteString ?? "")
                } else {
                    TextField("Address", text: $connection.baseURL)
                    if let addressProblem {
                        Text(addressProblem).font(.caption).foregroundStyle(.orange)
                    } else if addressChanged && hadKey {
                        Text("Enter the key again for the new address.").font(.caption).foregroundStyle(.orange)
                    }
                }
                if showsKey {
                    HStack {
                        SecureField("API key", text: $key, prompt: Text(hadKey && !keyRemoved && !addressChanged ? "Key saved" : "Paste the key"))
                        Button("Remove") { key = ""; keyRemoved = true }.disabled(!hadKey || keyRemoved)
                    }
                }
                HStack {
                    TextField("Default model", text: $connection.model)
                    Menu("Choose…") { ForEach(models, id: \.self) { id in Button(id) { connection.model = id } } }
                        .disabled(models.isEmpty).fixedSize()
                }
                HStack {
                    Button(testing ? "Testing…" : "Test connection") { Task { await test() } }
                        .disabled(testing || addressProblem != nil)
                    if let testResult { Text(testResult).font(.caption).foregroundStyle(.secondary) }
                }
            }
            .formStyle(.grouped)
            HStack {
                if !isNew {
                    Button("Delete", role: .destructive) { delete() }
                }
                Spacer()
                Button("Cancel") { dismiss() }.keyboardShortcut(.cancelAction)
                Button("Save", action: save).keyboardShortcut(.defaultAction)
                    .disabled(addressProblem != nil || connection.name.trimmingCharacters(in: .whitespaces).isEmpty)
            }
        }
        .padding(20)
        .frame(width: 520)
    }

    /// Lists models with the key typed here (or the saved one); generates nothing.
    private func test() async {
        testing = true
        defer { testing = false }
        guard let url = ConnectionAddress.baseURL(for: connection) else { return }
        let typed = key.trimmingCharacters(in: .whitespaces)
        let apiKey = typed.isEmpty && !keyRemoved && !addressChanged ? model.keychain.read(original.keychainAccount) : typed
        do {
            models = try await ModelCatalog.list(baseURL: url, apiKey: apiKey, anthropic: connection.kind.usesAnthropicAPI)
            testResult = models.count == 1 ? "Reachable · 1 model" : "Reachable · \(models.count) models"
        } catch {
            testResult = error.localizedDescription
        }
    }

    private func save() {
        let typed = key.trimmingCharacters(in: .whitespaces)
        do {
            if !typed.isEmpty {
                try model.keychain.write(typed, account: connection.keychainAccount)
            } else if keyRemoved || (addressChanged && hadKey) {
                try model.keychain.write(nil, account: connection.keychainAccount)
            }
        } catch {
            model.saveError = "Could not save the API key: \(error.localizedDescription)"
            return
        }
        connection.name = connection.name.trimmingCharacters(in: .whitespaces)
        connection.baseURL = connection.baseURL.trimmingCharacters(in: .whitespaces)
        if let index = model.settings.models.connections.firstIndex(where: { $0.id == connection.id }) {
            model.settings.models.connections[index] = connection
        } else {
            model.settings.models.connections.append(connection)
        }
        let saved = connection
        Task { await model.testConnection(saved) }
        dismiss()
    }

    /// The key goes with the connection. Rows that used it show "Choose a model".
    private func delete() {
        try? model.keychain.write(nil, account: original.keychainAccount)
        model.settings.models.connections.removeAll { $0.id == original.id }
        model.connectionStatus[original.id] = nil
        dismiss()
    }
}
```
