# Upload Audio/Video Implementation Plan

**Goal:** Transcribe selected audio/video files through the configured speech engine, one at a time, into durable, playable uploaded meetings.

**Architecture:** Decode bounded AVFoundation audio blocks into the existing durable PCM journal, then commit a transcribing manifest before running file-only meeting chunks. A Foundation-only queue owns admission; an app-lifetime controller owns URLs, security scopes, workers and source-specific recovery. Reuse the shared transcription router, meeting library, player, generation and exports without recording capture or upload Insights events.

**Tech Stack:** Swift tools 6.0, Swift 6.4, macOS 15.0, Foundation, AVFoundation, AudioToolbox, SwiftUI, AppKit, UniformTypeIdentifiers, Swift Testing; WhisperKit `exact: "1.1.1"`.

**Spec:** `docs/superpowers/specs/2026-10-15-upload-design.md`. Visual reference: `docs/design/proposal-upload.html`.

Execute each task as one commit. Code blocks below are additions or named replacements, not whole-file replacements unless stated. Preserve unrelated code. Line ranges refer to the repository read on 2026-10-15; locate the named declaration again if another task shifts lines. No implementation is performed by this document-authoring commit.

- `VoiceCore` imports Foundation only; Swift tools 6.0; macOS deployment floor 15.0.
- No new dependency; WhisperKit remains pinned `exact: "1.1.1"`; no ffmpeg or decoder subprocess.
- `MeetingManifest.currentSchemaVersion` stays `1`; explicitly decode missing source as `.recording`, missing original name/import date as nil; other decoding requirements remain unchanged.
- VoiceCore changes use failing tests first; run `swift test` before every implementation commit, including docs commits.
- Build the signed app only with `Scripts/build-app.sh`, using `Bugler Local Dev`; never change signing.
- API keys remain in `credentials.json`, directory mode `0700`, file mode `0600`; never settings or new Keychain entries.
- Accept exactly MP3, WAV, M4A, AIFF, CAF, MP4, MOV, M4V, case-insensitively; use the eight extension-derived UTTypes; reject directories without traversal.
- Journal PCM is signed Int16 little-endian, 16,000 Hz, mono; decoder blocks are Float32, at most 16,000 samples; final audio is 16 kHz mono AAC `audio.m4a`.
- Keep the journal until audio is durable AND the ready manifest is saved; transcript failure keeps the journal for Retry. Never reconstruct it from AAC.
- Keep original files unchanged, store only last path component as metadata, never persist source URLs; no microphone/system capture for uploads.
- Recognition uses `AppModel.transcriptionRouter`; network use follows only the user's configured Speech or Meeting notes server; secure-field paste rules remain unchanged.
- UI copy is plain English, no hedging, no emoji; exact §5 labels: `Upload`, `Upload a recording`, `Transcribed on this Mac with <speech model>`, `Transcribed by <server host> with <model>`, `Drop an audio or video file, or click to browse`, `MP3, WAV, M4A, AIFF, CAF, MP4, MOV, M4V`, `You can drop several files; they are transcribed one at a time. - Zero-length file → Task 5 `emptyAndVideoOnlyAreRejected`; Task 7 manual failure/next-job check verifies removal and queue advancement.

```swift
import Foundation

public enum UploadMediaKind: String, Sendable {
    case audio = "Audio"
    case video = "Video"
}

public enum UploadJobState: Sendable {
    case queued
    case reading
    case transcribing
    case ready
    case failed
    case cancelled

    public var isTerminal: Bool {
        self == .ready || self == .failed || self == .cancelled
    }
}

public struct UploadJob: Identifiable, Equatable, Sendable {
    public let id: UUID
    public var fileName: String
    public var kind: UploadMediaKind
    public var duration: TimeInterval
    public var byteSize: Int64
    public var meetingID: MeetingID?
    public var state: UploadJobState = .queued
    public var progress: Double?
    public var errorMessage: String?

    public init(id: UUID = UUID(), fileName: String, kind: UploadMediaKind, duration: TimeInterval, byteSize: Int64) {
        self.id = id
        self.fileName = fileName
        self.kind = kind
        self.duration = duration
        self.byteSize = byteSize
    }
}

public enum UploadEvent: Sendable {
    case progress(Double?)
    case committed(MeetingID)
    case ready
    case failed(String)
    case cleanupFailed
    case cancelled
}

public struct UploadQueue: Sendable {
    public private(set) var jobs: [UploadJob] = []

    public init() {}

    public var active: UploadJob? {
        jobs.first { $0.state == .reading || $0.state == .transcribing }
    }

    public mutating func append(_ batch: [UploadJob]) {
        for job in batch where job.state == .queued && !jobs.contains(where: { $0.id == job.id }) {
            jobs.append(job)
        }
    }

    public mutating func admit(meetingActive: Bool) -> UploadJob? {
        guard !meetingActive, active == nil,
              let index = jobs.firstIndex(where: { $0.state == .queued }) else { return nil }
        jobs[index].state = .reading
        jobs[index].progress = 0
        return jobs[index]
    }

    public mutating func apply(id: UUID, event: UploadEvent) {
        guard let index = jobs.firstIndex(where: { $0.id == id }), !jobs[index].state.isTerminal else { return }
        let state = jobs[index].state
        switch event {
        case .progress(let value) where state == .reading || state == .transcribing:
            jobs[index].progress = value.flatMap { $0.isFinite ? min(1, max(0, $0)) : nil }
        case .committed(let meetingID) where state == .reading:
            jobs[index].meetingID = meetingID
            jobs[index].state = .transcribing
            jobs[index].progress = 0
        case .ready where state == .transcribing:
            jobs[index].state = .ready
            jobs[index].progress = 1
        case .failed(let message) where state == .reading || state == .transcribing:
            jobs[index].state = .failed
            jobs[index].errorMessage = message
        case .cleanupFailed:
            jobs[index].state = .failed
            jobs[index].errorMessage = "Could not remove the cancelled upload."
        case .cancelled:
            jobs[index].state = .cancelled
            jobs[index].progress = nil
        default:
            break
        }
    }

    public mutating func clearFinished() {
        jobs.removeAll { $0.state.isTerminal }
    }

    public mutating func reconcile(existing: Set<MeetingID>) {
        for index in jobs.indices where jobs[index].state.isTerminal {
            if let id = jobs[index].meetingID, !existing.contains(id) {
                jobs[index].meetingID = nil
            }
        }
    }
}
```

```swift
import Foundation

public enum UploadFailure: Error, LocalizedError, Equatable, Sendable {
    case noAudio
    case unreadable
    case diskFull
    case transcript
    case audioSave
    case missingJournal
    case cleanup

    public var errorDescription: String? {
        switch self {
        case .noAudio: "This file has no audio."
        case .unreadable: "BugVoice can't read this file. Use MP3, WAV, M4A, AIFF, CAF, MP4, MOV or M4V."
        case .diskFull: "Not enough disk space to transcribe this file."
        case .transcript: "Transcript could not be completed. The audio is saved; use Retry transcript in Meetings."
        case .audioSave: "Could not save the audio. Retry transcript in Meetings."
        case .missingJournal: "The transcript cannot be retried. The audio is saved."
        case .cleanup: "Could not remove the cancelled upload."
        }
    }

    public static func isDiskFull(_ error: any Error) -> Bool {
        if let failure = error as? UploadFailure, failure == .diskFull { return true }
        let ns = error as NSError
        if ns.domain == NSCocoaErrorDomain && ns.code == CocoaError.fileWriteOutOfSpace.rawValue { return true }
        if ns.domain == NSPOSIXErrorDomain && ns.code == 28 { return true }
        if let underlying = ns.userInfo[NSUnderlyingErrorKey] as? NSError {
            return isDiskFull(underlying)
        }
        return false
    }
}
```

```swift
    public var kind: UploadMediaKind

public struct UploadPCMStream: Sendable {
    private let totalSamples: Int
    private let sink: @Sendable ([Float], TimeInterval) async throws -> Void
    private let checkpoint: @Sendable () async throws -> Void
    private var position = 0
    private var sinceCheckpoint = 0

    public init(
        totalSamples: Int,
        sink: @escaping @Sendable ([Float], TimeInterval) async throws -> Void,
        checkpoint: @escaping @Sendable () async throws -> Void
    ) {
        self.totalSamples = totalSamples
        self.sink = sink
        self.checkpoint = checkpoint
    }

    public mutating func push(_ samples: [Float], at start: Int) async throws {
        while position < min(start, totalSamples) {
            try await emit(Array(repeating: 0, count: min(16_000 - sinceCheckpoint, min(start, totalSamples) - position)))
        }
        var index = max(0, position - start)
        while index < samples.count, position < totalSamples {
            let count = min(16_000 - sinceCheckpoint, samples.count - index, totalSamples - position)
            try await emit(Array(samples[index..<(index + count)]))
            index += count
        }
    }

    public mutating func finish() async throws {
        while position < totalSamples {
            try await emit(Array(repeating: 0, count: min(16_000 - sinceCheckpoint, totalSamples - position)))
        }
        try await checkpoint()
    }

    private mutating func emit(_ samples: [Float]) async throws {
        try Task.checkCancellation()
        try await sink(samples, Double(position) / 16_000)
        position += samples.count
        sinceCheckpoint += samples.count
        if sinceCheckpoint >= 16_000 {
            try await checkpoint()
            sinceCheckpoint = 0
        }
    }
}

            throw CancellationError()
        } catch let failure as UploadFailure {
            throw failure
        } catch {

        metadata: UploadMediaMetadata,

        progress: @escaping @Sendable (Double?) async -> Void

            tracks = try await asset.loadTracks(withMediaType: .audio)

            AVFormatIDKey: kAudioFormatLinearPCM,
            AVSampleRateKey: 16_000,
            AVNumberOfChannelsKey: 1,
            AVLinearPCMBitDepthKey: 32,
            AVLinearPCMIsFloatKey: true,
            AVLinearPCMIsBigEndianKey: false,
            AVLinearPCMIsNonInterleaved: false

            let range = try await track.load(.timeRange)
            expectedAudioEnd = max(expectedAudioEnd, CMTimeRangeGetEnd(range).seconds)

        try writer.start(track: .file)
        defer { writer.close() }

            writer.append(samples, track: .file, offset: offset)
        }, checkpoint: {
            _ = try writer.checkpoint()
            try await boundary()

        defer { reader.cancelReading() }

            defer { withExtendedLifetime(buffer) {} }
            try Task.checkCancellation()
            let time = CMSampleBufferGetPresentationTimeStamp(buffer).seconds
            guard time.isFinite, let block = CMSampleBufferGetDataBuffer(buffer) else { throw UploadFailure.unreadable }
            let count = CMSampleBufferGetNumSamples(buffer)
            var pointer: UnsafeMutablePointer<Int8>?

            var length = 0

            while offset < count {
                let size = min(16_000, count - offset)
                let samples = Array(UnsafeBufferPointer(start: floats.advanced(by: offset), count: size))
                try await stream.push(samples, at: start + offset)
                offset += size
                await progress(min(1, max(0, Double(start + offset) / Double(total))))
            }
            decodedEnd = max(decodedEnd, time + Double(count) / 16_000)
        }
        guard reader.status == .completed,
              decodedEnd + 0.1 >= min(metadata.duration, expectedAudioEnd) else {
            throw UploadFailure.unreadable
        }
        await progress(nil)
        try await stream.finish()
        _ = try writer.checkpoint()
        try Task.checkCancellation()
    }
```

```swift
enum MeetingAACWriter {
    static func write(
        sampleCount: Int,
        readBlock: (_ offset: Int, _ count: Int) throws -> [Float],
        to destination: URL
    ) throws {
        try FileManager.default.createDirectory(at: destination.deletingLastPathComponent(), withIntermediateDirectories: true)
        let temporary = destination.deletingLastPathComponent()
            .appendingPathComponent(".\(destination.deletingPathExtension().lastPathComponent).\(UUID().uuidString).tmp.m4a")
        do {
            let settings: [String: Any] = [
                AVFormatIDKey: kAudioFormatMPEG4AAC,
                AVSampleRateKey: 16_000,
                AVNumberOfChannelsKey: 1
            ]
            let outputFormat = AVAudioFormat(commonFormat: .pcmFormatFloat32, sampleRate: 16_000, channels: 1, interleaved: false)!
            let file = try AVAudioFile(forWriting: temporary, settings: settings, commonFormat: .pcmFormatFloat32, interleaved: false)
            let finalFrameCount = max(1, sampleCount)
            var offset = 0
            while offset < finalFrameCount {
                let count = sampleCount == 0 ? 1 : min(16_000, sampleCount - offset)
                let samples = sampleCount == 0 ? [] : try readBlock(offset, count)
                guard samples.count == (sampleCount == 0 ? 0 : count),
                      let buffer = AVAudioPCMBuffer(pcmFormat: outputFormat, frameCapacity: AVAudioFrameCount(count)),
                      let channel = buffer.floatChannelData?[0] else {
                    throw MeetingAudioFinalizationError.audioFile("Could not read or allocate an AAC input block")
                }
                buffer.frameLength = AVAudioFrameCount(count)
                if samples.isEmpty { channel[0] = 0 }
                else { samples.withUnsafeBufferPointer { channel.update(from: $0.baseAddress!, count: samples.count) } }
                try file.write(from: buffer)
                offset += count
            }
            file.close()
            let handle = try FileHandle(forWritingTo: temporary)
            try handle.synchronize()
            try handle.close()
            guard rename(temporary.path, destination.path) == 0 else {
                throw NSError(domain: NSPOSIXErrorDomain, code: Int(errno))
            }
            try syncDirectory(destination.deletingLastPathComponent())
        } catch {
            try? FileManager.default.removeItem(at: temporary)
            throw error
        }
    }

    static func syncDirectory(_ url: URL) throws {
        let descriptor = open(url.path, O_RDONLY)
        guard descriptor >= 0 else { throw NSError(domain: NSPOSIXErrorDomain, code: Int(errno)) }
        let status = fsync(descriptor)
        let error = status == 0 ? nil : NSError(domain: NSPOSIXErrorDomain, code: Int(errno))
        _ = close(descriptor)
        if let error { throw error }
    }
}
```

```swift
actor UploadTranscriptionScheduler: MeetingTranscriptionScheduler {
    private let writer: MeetingPCMJournalWriter
    private let transcriber: any Transcriber
    private let language: SpeechLanguage
    private let gate: @Sendable () async throws -> Void
    private weak var session: MeetingSession?
    private var cancelled = false
    private var running = false
    private var waiters: [CheckedContinuation<Void, Never>] = []

    init(writer: MeetingPCMJournalWriter, transcriber: any Transcriber, language: SpeechLanguage, gate: @escaping @Sendable () async throws -> Void) {
        self.writer = writer
        self.transcriber = transcriber
        self.language = language
        self.gate = gate
    }

    func bindSession(_ session: MeetingSession) {
        self.session = session
    }

    func cancelQueued(meetingID: MeetingID) async {
        cancelled = true
    }

    func waitForInFlight(meetingID: MeetingID) async {
        guard running else { return }
        await withCheckedContinuation { waiters.append($0) }
    }

    func transcribeFinal(
        _ manifest: MeetingManifest,
        chunkSeconds: TimeInterval,
        progress: @escaping @Sendable (Double) async -> Void
    ) async throws -> [MeetingChunkProcessingResult] {
        guard manifest.source == .upload, !running, let session else { throw MeetingSessionError.invalidState }
        cancelled = false
        running = true
        defer {
            running = false
            let pending = waiters
            waiters.removeAll()
            for waiter in pending { waiter.resume() }
        }
        let checkpoint = try writer.storedCheckpoint()
        let count = checkpoint.track(.file)?.sampleCount ?? 0
        guard count == Int((manifest.duration * 16_000).rounded()), count > 0 else { throw UploadFailure.missingJournal }
        let ranges = MeetingChunkPlanner.plan(sampleCount: count, chunkSeconds: chunkSeconds)
        let gated = UploadGatedTranscriber(base: transcriber, wait: gate)
        var results: [MeetingChunkProcessingResult] = []
        for range in ranges {
            try Task.checkCancellation()
            guard !cancelled else { throw CancellationError() }
            try await gate()
            let result: MeetingChunkProcessingResult
            if let record = UploadChunkPlan.reusable(manifest.chunks, range: range) {
                // Reconstruct from the chosen original attempt, not already merged text.
                // A merged boundary can have lost its prefix; replaying that would trim twice.
                let chosen = record.chosenAttempt.flatMap { index in
                    record.attempts.indices.contains(index) ? record.attempts[index] : nil
                }
                let segments = chosen?.segments.enumerated().map { index, segment in
                    MeetingSegment(
                        id: "\(record.id)-\(index)",
                        start: record.offset + segment.start,
                        end: record.offset + segment.end,
                        text: segment.text,
                        track: .file,
                        sourceChunkID: record.id
                    )
                } ?? manifest.segments.filter { $0.sourceChunkID == record.id }
                result = MeetingChunkProcessingResult(record: record, segments: segments)
            } else {
                let samples = try await writer.committedSamplesAsync(for: .file, range: range)
                guard samples.count == range.count else { throw UploadFailure.missingJournal }
                var processed = try await MeetingChunkProcessor().process(
                    id: UploadChunkPlan.id(for: range), track: .file,
                    offset: Double(range.lowerBound) / 16_000,
                    clip: AudioClip(samples: samples), configuredLanguage: language, transcriber: gated
                )
                for index in processed.segments.indices {
                    processed.segments[index].id = "\(processed.record.id)-\(index)"
                }
                result = processed
            }
            try Task.checkCancellation()
            guard !cancelled else { throw CancellationError() }
            try await session.persistFinal(result)
            results.append(result)
            await progress(Double(results.count) / Double(ranges.count))
            if result.record.state == .failed { throw UploadFailure.transcript }
        }
        return results
    }
}
```

```swift
import AppKit
import Foundation
import Observation
import VoiceCore

@MainActor
@Observable
final class UploadController {
    struct Draft: Identifiable, Sendable {
        let id = UUID()
        let url: URL
        let metadata: UploadMediaMetadata
        let scoped: Bool
    }

    private unowned let model: AppModel
    private(set) var queue = UploadQueue()
    private(set) var drafts: [Draft] = []
    private(set) var draftError: String?
    private(set) var errorFileName: String?
    private(set) var probing = false
    private(set) var sleeping = false
    private(set) var accepting = false
    private(set) var waitingForMeeting = false
    @ObservationIgnored private var inputs: [UUID: Draft] = [:]
    @ObservationIgnored private var restored: [UUID: MeetingManifest] = [:]
    @ObservationIgnored private var worker: Task<Void, Never>?
    @ObservationIgnored private var decoder: Task<Void, Error>?
    @ObservationIgnored private var session: MeetingSession?
    @ObservationIgnored private var scheduler: UploadTranscriptionScheduler?
    @ObservationIgnored private var activeID: UUID?
    @ObservationIgnored private var activeDirectory: URL?
    @ObservationIgnored private var progressMonitor: Task<Void, Never>?
    private var stopping = false
    private var selectionVersion = 0

    init(model: AppModel) {
        self.model = model
        progressMonitor = Task { [weak self] in
            while !Task.isCancelled {
                try? await Task.sleep(for: .milliseconds(200))
                guard !Task.isCancelled else { return }
                await self?.refreshProgress()
            }
        }
    }

    var isBusy: Bool { worker != nil || !queue.jobs.filter { !$0.state.isTerminal }.isEmpty }
    var active: UploadJob? { queue.active }

    func choose(_ urls: [URL]) async {
        guard accepting else { return }
        clearDraft()
        selectionVersion += 1
        let version = selectionVersion
        probing = true
        defer { if version == selectionVersion { probing = false } }
        var selected: [Draft] = []
        for url in urls {
            let scoped = url.startAccessingSecurityScopedResource()
            do {
                let metadata = try await Task.detached(priority: .utility) {
                    try await MediaAudioDecoder.probe(url)
                }.value
                guard version == selectionVersion else {
                    if scoped { url.stopAccessingSecurityScopedResource() }
                    for item in selected { release(item) }
                    return
                }
                selected.append(Draft(url: url, metadata: metadata, scoped: scoped))
            } catch {
                if scoped { url.stopAccessingSecurityScopedResource() }
                for item in selected { release(item) }
                guard version == selectionVersion else { return }
                errorFileName = url.lastPathComponent
                draftError = (error as? UploadFailure ?? .unreadable).localizedDescription
                return
            }
        }
        drafts = selected
    }

    func clearDraft() {
        selectionVersion += 1
        for item in drafts { release(item) }
        drafts.removeAll()
        draftError = nil
        errorFileName = nil
        probing = false
    }

    func submit() {
        guard accepting, !probing else { return }
        let jobs = drafts.map { draft in
            let job = UploadJob(fileName: draft.metadata.fileName, kind: draft.metadata.kind, duration: draft.metadata.duration, byteSize: draft.metadata.byteSize)
            inputs[job.id] = draft
            return job
        }
        drafts.removeAll()
        queue.append(jobs)
        pump()
    }

    func clearFinished() {
        queue.clearFinished()
    }

    func restore(_ manifests: [MeetingManifest]) {
        for manifest in manifests.sorted(by: { $0.id.rawValue < $1.id.rawValue }) {
            appendRecovery(manifest)
        }
        accepting = true
        pump()
    }

    func retry(_ meeting: MeetingManifest) {
        guard meeting.source == .upload, meeting.state == .transcribing,
              !queue.jobs.contains(where: { $0.meetingID == meeting.id && !$0.state.isTerminal }) else { return }
        appendRecovery(meeting)
        pump()
    }

    private func appendRecovery(_ meeting: MeetingManifest) {
        var job = UploadJob(fileName: meeting.sourceFileName ?? meeting.title, kind: .audio, duration: meeting.duration, byteSize: 0)
        job.meetingID = meeting.id
        restored[job.id] = meeting
        queue.append([job])
    }

    func reconcile(_ manifests: [MeetingManifest]) {
        queue.reconcile(existing: Set(manifests.map(\.id)))
    }

    func open(_ id: MeetingID) {
        model.selectedMeetingID = id
        model.openMainWindow(.meetings)
    }

    func setSleeping(_ value: Bool) {
        sleeping = value
        if !value { pump() }
    }

    private func waitForReadingBoundary() async throws {
        while sleeping {
            try Task.checkCancellation()
            try await Task.sleep(for: .milliseconds(100))
        }
        try Task.checkCancellation()
    }

    private func waitForRecognition() async throws {
        while sleeping || model.meetings?.isActive == true {
            waitingForMeeting = model.meetings?.isActive == true
            try Task.checkCancellation()
            try await Task.sleep(for: .milliseconds(100))
        }
        waitingForMeeting = false
        try Task.checkCancellation()
    }

    private func pump() {
        guard worker == nil, accepting, !stopping else { return }
        worker = Task { [weak self] in
            guard let self else { return }
            defer {
                self.worker = nil
                self.activeID = nil
                self.activeDirectory = nil
                self.session = nil
                self.scheduler = nil
            }
            while !Task.isCancelled && !self.stopping {
                // Keep queued jobs waiting through recording, pause and final pass.
                if self.sleeping || self.model.meetings?.isActive == true {
                    do { try await self.waitForRecognition() } catch { return }
                }
                guard let job = self.queue.admit(meetingActive: false) else { return }
                await self.run(job)
            }
        }
    }

    private func run(_ job: UploadJob) async {
        let saved = restored.removeValue(forKey: job.id)
        let id = saved?.id ?? MeetingID.new()
        let directory = model.paths.meetings.appendingPathComponent(id.rawValue, isDirectory: true)
        activeID = job.id
        activeDirectory = directory
        queue.attach(id: job.id, meetingID: id)
        let writer = MeetingPCMJournalWriter(directory: directory.appendingPathComponent(".capture"))
        let scheduler = UploadTranscriptionScheduler(
            writer: writer, transcriber: model.transcriptionRouter,
            language: model.settings.speech.language,
            gate: { [weak self] in
                guard let self else { throw CancellationError() }
                try await self.waitForRecognition()
            }
        )
        let session = MeetingSession(
            store: MeetingStore(root: model.paths.meetings),
            audio: UploadAudioSourceControl(writer: writer),
            scheduler: scheduler,
            finalizer: UploadAudioFinalizer(meetingDirectory: directory),
            meetingID: id
        )
        self.scheduler = scheduler
        self.session = session
        await scheduler.bindSession(session)
        do {
            if let saved {
                queue.apply(id: job.id, event: .committed(saved.id))
                _ = try await session.recover(id: saved.id)
            } else {
                guard let draft = inputs[job.id] else { throw UploadFailure.unreadable }
                let now = Date()
                let startedAt = draft.metadata.creationDate ?? now
                var manifest = MeetingManifest(
                    id: id,
                    title: (draft.metadata.fileName as NSString).deletingPathExtension,
                    startedAt: startedAt,
                    endedAt: startedAt.addingTimeInterval(draft.metadata.duration),
                    duration: draft.metadata.duration,
                    state: .interrupted,
                    source: .upload,
                    sourceFileName: draft.metadata.fileName,
                    importedAt: now
                )
                try await MeetingStore(root: model.paths.meetings).save(manifest)
                let decoding = Task.detached(priority: .utility) { [weak self] in
                    try await MediaAudioDecoder.decode(draft.url, metadata: draft.metadata, writer: writer, boundary: {
                        guard let self else { throw CancellationError() }
                        try await self.waitForReadingBoundary()
                    }, progress: { value in
                        await self?.readingProgress(value, id: job.id)
                    })
                }
                decoder = decoding
                try await withTaskCancellationHandler {
                    try await decoding.value
                } onCancel: {
                    decoding.cancel()
                }
                decoder = nil
                releaseInput(job.id)
                try Task.checkCancellation()
                manifest.state = .transcribing
                // Writer is closed, EOF checkpoint succeeded: atomically commit before speech.
                try await MeetingStore(root: model.paths.meetings).save(manifest)
                queue.apply(id: job.id, event: .committed(id))
                _ = try await session.transcribeUpload(manifest)
            }
            try Task.checkCancellation()
            guard !stopping else { return }
            queue.apply(id: job.id, event: .ready)
        } catch {
            decoder = nil
            releaseInput(job.id)
            if stopping || error is CancellationError { return }
            let failure: UploadFailure
            if UploadFailure.isDiskFull(error) {
                failure = .diskFull
                await removeFailedImport(directory)
            } else if let upload = error as? UploadFailure, upload == .transcript || upload == .audioSave {
                failure = upload
            } else if saved != nil, error as? UploadFailure == .missingJournal {
                failure = .missingJournal
                var failed = saved!
                failed.state = .failed
                failed.transcriptionError = failure.localizedDescription
                do { try await MeetingStore(root: model.paths.meetings).save(failed) }
                catch { model.saveError = error.localizedDescription }
            } else {
                failure = (error as? UploadFailure) == .noAudio ? .noAudio : .unreadable
                await removeFailedImport(directory)
            }
            queue.apply(id: job.id, event: .failed(failure.localizedDescription))
        }
        await model.reloadMeetings()
    }

    private func readingProgress(_ value: Double?, id: UUID) {
        guard !stopping, activeID == id else { return }
        queue.apply(id: id, event: .progress(value))
    }

    func refreshProgress() async {
        guard !stopping, let id = activeID, queue.active?.state == .transcribing,
              let manifest = await session?.snapshot() else { return }
        queue.apply(id: id, event: .progress(manifest.transcriptProgress))
    }

    func cancel(_ id: UUID) async {
        guard let job = queue.jobs.first(where: { $0.id == id }), !job.state.isTerminal else { return }
        if id != activeID {
            if let meeting = restored[id] {
                let directory = model.paths.meetings.appendingPathComponent(meeting.id.rawValue)
                do {
                    try markForRemoval(directory)
                    try FileManager.default.removeItem(at: directory)
                } catch {
                    queue.apply(id: id, event: .cleanupFailed)
                    return
                }
            }
            releaseInput(id)
            restored.removeValue(forKey: id)
            queue.apply(id: id, event: .cancelled)
            await model.reloadMeetings()
            return
        }
        guard let directory = activeDirectory else { return }
        do {
            try markForRemoval(directory)
        } catch {
            model.saveError = UploadFailure.cleanup.localizedDescription
            return
        }
        stopping = true
        await session?.invalidateUpload()
        worker?.cancel()
        decoder?.cancel()
        let draining = worker
        await draining?.value
        do {
            try FileManager.default.removeItem(at: directory)
            queue.apply(id: id, event: .cancelled)
        } catch {
            queue.apply(id: id, event: .cleanupFailed)
        }
        releaseInput(id)
        stopping = false
        await model.reloadMeetings()
        pump()
    }

    func quiesceForQuit() async {
        accepting = false
        stopping = true
        progressMonitor?.cancel()
        progressMonitor = nil
        clearDraft()
        worker?.cancel()
        decoder?.cancel()
        let draining = worker
        await draining?.value
        for id in Array(inputs.keys) { releaseInput(id) }
        // No marker, no deletion: interrupted reading is discardable; transcribing is resumable.
    }

    private func release(_ draft: Draft) {
        if draft.scoped { draft.url.stopAccessingSecurityScopedResource() }
    }

    private func releaseInput(_ id: UUID) {
        if let draft = inputs.removeValue(forKey: id) { release(draft) }
    }

    private func markForRemoval(_ directory: URL) throws {
        try FileManager.default.createDirectory(at: directory, withIntermediateDirectories: true)
        let marker = directory.appendingPathComponent(".upload-cancelled")
        try Data().write(to: marker)
        let handle = try FileHandle(forWritingTo: marker)
        try handle.synchronize()
        try handle.close()
        // Directory sync uses the same macOS durability primitive as AAC replacement.
        try UploadCancellationMarker.sync(directory)
    }

    private func removeFailedImport(_ directory: URL) async {
        do {
            try markForRemoval(directory)
            try FileManager.default.removeItem(at: directory)
        } catch {
            model.saveError = UploadFailure.cleanup.localizedDescription
        }
    }
}
```

```swift
import Foundation

public struct UploadRecoveryPlan: Sendable {
    public let discard: [MeetingID]
    public let resume: [MeetingManifest]
    public let ready: [MeetingManifest]

    public init(_ manifests: [MeetingManifest]) {
```

```swift
private func recoverUnfinishedMeetings() async {
    do {
        // Scan marker paths BEFORE decoding manifests, even if a manifest is corrupt.
        let directories = try FileManager.default.contentsOfDirectory(
            at: paths.meetings,
            includingPropertiesForKeys: [.isDirectoryKey],
            options: [.skipsHiddenFiles]
        )
        var blocked: Set<MeetingID> = []

            guard (try? directory.resourceValues(forKeys: [.isDirectoryKey]).isDirectory) == true else { continue }
            let marker = directory.appendingPathComponent(".upload-cancelled")
            if FileManager.default.fileExists(atPath: marker.path) {
                do {
                    try FileManager.default.removeItem(at: directory)
                } catch {
                    blocked.insert(MeetingID(rawValue: directory.lastPathComponent))
                    saveError = UploadFailure.cleanup.localizedDescription
                }
            }
        }
        // Load individually: a marked folder is never loaded; one corrupt folder cannot stop all recovery.
        var all: [MeetingManifest] = []
        for directory in directories {
            let id = MeetingID(rawValue: directory.lastPathComponent)
            guard !blocked.contains(id),
                  FileManager.default.fileExists(atPath: directory.appendingPathComponent("meeting.json").path) else { continue }
            do {
                all.append(try await meetingStore.load(id: id))
            } catch {
                saveError = "Could not recover meetings: \(error.localizedDescription)"
            }
        }
        let plan = UploadRecoveryPlan(all)
        for id in plan.discard {
            let directory = paths.meetings.appendingPathComponent(id.rawValue)
            // Mark before removing, so an unsuccessful cleanup remains excluded on the next launch.
            let marker = directory.appendingPathComponent(".upload-cancelled")
            do {
                try Data().write(to: marker)
                let handle = try FileHandle(forWritingTo: marker)
                try handle.synchronize()
                try handle.close()
                try UploadCancellationMarker.sync(directory)
                try await meetingStore.delete(id: id)
            } catch {
                blocked.insert(id)
                saveError = UploadFailure.cleanup.localizedDescription
            }
        }
        for meeting in plan.ready {
            let directory = paths.meetings.appendingPathComponent(meeting.id.rawValue)
            let audio = directory.appendingPathComponent("audio.m4a")
            // Validate the durable replacement before discarding a redundant journal.
            if let metadata = try? await MediaAudioDecoder.probe(audio),
               abs(metadata.duration - meeting.duration) < 0.2 {
                do {
                    try await UploadAudioFinalizer(meetingDirectory: directory).cleanupCapture()
                } catch {
                    saveError = "Could not remove redundant upload audio: \(error.localizedDescription)"
                }
            }
        }
        for var meeting in all where meeting.source == .recording && (meeting.state == .recording || meeting.state == .paused) {
            meeting.state = .interrupted
            meeting.endedAt = meeting.endedAt ?? Date()
```

```swift
private nonisolated static func readWaveform(_ url: URL) -> [Float] {
    guard let file = try? AVAudioFile(forReading: url) else { return [] }
    let frames = Int(file.length)
    guard frames > 0 else { return [] }
    let bucketCount = 180
    var bars: [Float] = []
    guard let buffer = AVAudioPCMBuffer(pcmFormat: file.processingFormat, frameCapacity: 16_000) else { return [] }
    for bucket in 0..<bucketCount {
        let start = bucket * frames / bucketCount
        let end = (bucket + 1) * frames / bucketCount
        file.framePosition = AVAudioFramePosition(start)
        var remaining = end - start
        var peak: Float = 0
        while remaining > 0 {
            if Task.isCancelled { return [] }
            let count = min(16_000, remaining)
            do {
                try file.read(into: buffer, frameCount: AVAudioFrameCount(count))
            } catch {
                return []
            }
            let samples = Int(buffer.frameLength)
            guard samples > 0 else { return [] }
            if let channel = buffer.floatChannelData?[0] {
                for index in 0..<samples { peak = max(peak, abs(channel[index])) }
            } else if let channel = buffer.int16ChannelData?[0] {
                for index in 0..<samples { peak = max(peak, abs(Float(channel[index]) / 32_768)) }
            }
            remaining -= samples
        }
        bars.append(min(1, peak))
    }
    return bars
}
```
